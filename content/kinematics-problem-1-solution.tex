\section*{Solution --- Accelerating Car}

\textbf{Given:} $u = 0\ \text{m/s}$, \quad $a = 2\ \text{m/s}^{2}$,
\quad $t = 5\ \text{s}$.

\subsection*{Part 1 --- Final velocity}
We know $u, a, t$ and want $v$:
\begin{equation}
  v = u + at = 0 + (2)(5) = 10\ \text{m/s}
\end{equation}

\subsection*{Part 2 --- Distance travelled}
\begin{equation}
  s = ut + \tfrac{1}{2}at^{2}
    = (0)(5) + \tfrac{1}{2}(2)(5)^{2} = 25\ \text{m}
\end{equation}

\subsection*{Check}
A second, independent equation must agree:
\begin{equation}
  s = \frac{(u+v)}{2}\,t = \frac{(0+10)}{2}(5) = 25\ \text{m}\;\checkmark
\end{equation}
